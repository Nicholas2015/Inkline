cmake_minimum_required(VERSION 3.21)
project(Inkline VERSION 0.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent Network Core5Compat)
find_path(QSCI_INCLUDE_DIR Qsci/qsciscintilla.h PATH_SUFFIXES include/qt6 include/Qt6 include REQUIRED)
find_library(QSCI_LIBRARY NAMES qscintilla2_qt6 qscintilla2-qt6 REQUIRED)
add_library(QScintilla::Qt6 UNKNOWN IMPORTED)
set_target_properties(QScintilla::Qt6 PROPERTIES
  IMPORTED_LOCATION "${QSCI_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${QSCI_INCLUDE_DIR}")
if(WIN32)
  target_compile_definitions(QScintilla::Qt6 INTERFACE QSCINTILLA_DLL)
endif()

add_subdirectory(third_party)

add_library(inkcore src/document.cpp src/logfile.cpp src/configtools.cpp)
target_include_directories(inkcore PRIVATE third_party/tomlplusplus)
target_link_libraries(inkcore PRIVATE inkconfigyaml)
target_include_directories(inkcore PUBLIC src)
target_link_libraries(inkcore PUBLIC Qt6::Core Qt6::Core5Compat)
add_library(inksearchtask STATIC src/searchtask.cpp)
target_include_directories(inksearchtask PUBLIC src)
target_link_libraries(inksearchtask PUBLIC Qt6::Core)

add_library(inkprovider STATIC src/providerservice.cpp)
target_include_directories(inkprovider PUBLIC src)
target_link_libraries(inkprovider PUBLIC Qt6::Network)

add_library(inkcode src/codetools.cpp src/languagetools.cpp)
target_include_directories(inkcode PUBLIC src)
target_link_libraries(inkcode PUBLIC Qt6::Concurrent Qt6::Widgets Qt6::Network QScintilla::Qt6 PRIVATE inkprovider)
add_library(inklog STATIC src/logview.cpp src/logscan.cpp)
target_link_libraries(inklog PUBLIC inkcore Qt6::Widgets Qt6::Concurrent)
qt_add_executable(Inkline src/main.cpp src/app/mainwindow.cpp src/app/windowmenus.cpp
  src/editcommands.cpp src/viewcommands.cpp src/app/windowfiles.cpp src/app/windowsession.cpp
  src/app/verifyinstall.cpp src/searchtools.cpp src/searchworker.cpp resources/icons.qrc)
target_link_libraries(Inkline PRIVATE inkcode inklog inksearchtask)
set_target_properties(Inkline PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER org.inkedit.native MACOSX_BUNDLE_ICON_FILE Inkline.icns)
if(WIN32)
 enable_language(RC)
 target_sources(Inkline PRIVATE resources/windows.rc)
elseif(APPLE)
 set_target_properties(Inkline PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/resources/Info.plist.in")
 set_source_files_properties(resources/Inkline.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
 target_sources(Inkline PRIVATE resources/Inkline.icns)
 foreach(pkg IN ITEMS libyaml tomlplusplus)
   set(license_file "third_party/${pkg}/LICENSE")
   set_source_files_properties("${license_file}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/licenses/${pkg}")
   target_sources(Inkline PRIVATE "${license_file}")
 endforeach()
endif()
install(TARGETS Inkline BUNDLE DESTINATION . RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
