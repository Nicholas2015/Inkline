cmake_minimum_required(VERSION 3.21)
project(Inkline VERSION 0.2 LANGUAGES CXX)
if(WIN32)
 add_compile_definitions(QSCINTILLA_DLL)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent Network Core5Compat)
find_path(QSCI_INCLUDE_DIR Qsci/qsciscintilla.h PATH_SUFFIXES include/qt6 include/Qt6 include REQUIRED)
find_library(QSCI_LIBRARY NAMES qscintilla2_qt6 qscintilla2-qt6 REQUIRED)
add_library(inkcore src/document.cpp src/logfile.cpp)
target_include_directories(inkcore PUBLIC src)
target_link_libraries(inkcore PUBLIC Qt6::Core Qt6::Core5Compat)
add_library(inkcode src/codetools.cpp src/languagetools.cpp)
target_include_directories(inkcode PUBLIC src ${QSCI_INCLUDE_DIR})
target_link_libraries(inkcode PUBLIC Qt6::Concurrent Qt6::Widgets Qt6::Network ${QSCI_LIBRARY})
qt_add_executable(Inkline src/main.cpp src/logview.cpp src/logscan.cpp src/searchtools.cpp src/searchworker.cpp resources/icons.qrc)
target_link_libraries(Inkline PRIVATE inkcode inkcore Qt6::Widgets Qt6::Network Qt6::Concurrent ${QSCI_LIBRARY})
target_include_directories(Inkline PRIVATE ${QSCI_INCLUDE_DIR})
set_target_properties(Inkline PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER org.inkedit.native MACOSX_BUNDLE_ICON_FILE Inkline.icns)
if(WIN32)
 enable_language(RC)
 target_sources(Inkline PRIVATE resources/windows.rc)
elseif(APPLE)
 set_target_properties(Inkline PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/resources/Info.plist.in")
 set_source_files_properties(resources/Inkline.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
 target_sources(Inkline PRIVATE resources/Inkline.icns)
endif()
install(TARGETS Inkline BUNDLE DESTINATION . RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
 add_executable(inkcheck tests/check.cpp)
 target_link_libraries(inkcheck PRIVATE inkcore)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 add_executable(inklogcheck tests/logui.cpp src/logview.cpp src/logscan.cpp)
 target_link_libraries(inklogcheck PRIVATE inkcore Qt6::Concurrent Qt6::Widgets Qt6::Test)
 add_test(NAME log_ui COMMAND inklogcheck)
 set_tests_properties(log_ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
 add_executable(inkcodecheck tests/codecheck.cpp)
 target_link_libraries(inkcodecheck PRIVATE inkcode)
 add_test(NAME code_tools COMMAND inkcodecheck)
 set_tests_properties(code_tools PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
 add_test(NAME format_search COMMAND inkcheck)
 add_test(NAME desktop_roundtrip COMMAND Inkline --verify-install "${CMAKE_CURRENT_BINARY_DIR}/desktop-check.json")
 set_tests_properties(desktop_roundtrip PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
endif()
